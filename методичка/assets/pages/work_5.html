Плавная прокрутка к секциям

В прошлой лабораторной работе прокрутка была реализована с помощью CSS scroll-behavior: smooth. Реализуем теперь ее с помощью JavaScript. Такая прокрутка будет больше контролироваться, можно настроить скорость, отступы и другие параметры.

document.querySelectorAll('nav a').forEach(link => {
    link.addEventListener('click', (e) => {
        e.preventDefault(); // Отменяем стандартное поведение
        const targetId = link.getAttribute('href');
        const targetSection = document.querySelector(targetId);
        
        // Плавная прокрутка
        targetSection.scrollIntoView({
            behavior: 'smooth',
            block: 'start'
        });
    });
});

dwindow.addEventListener('scroll', () => {
    document.querySelectorAll('section').forEach(section => {
        const sectionTop = section.getBoundingClientRect().top;
        if (sectionTop < window.innerHeight - 100) {
            section.classList.add('visible');
        }
    });
});

Темная тема

document.getElementById('theme-toggle').addEventListener('click', () => {
    document.body.classList.toggle('dark-theme');
    const btn = document.getElementById('theme-toggle');
    btn.textContent = document.body.classList.contains('dark-theme') 
        ? 'Светлая тема' 
        : 'Тёмная тема';
});
